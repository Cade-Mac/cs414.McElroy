#include <iostream>

void f(int y) {
    int x = y + 1;
    for (int i = 0; i < 3; i++) {
        int temp = i * 2 + x;
        std::cout << i << " " << temp << " " << x << std::endl;
    }
}

int main(int argc, char *argv[]) {
    int x = 2;
    f(x);
    std::cout << x << std::endl;
    return 0;
}


OUTLINE:
.
└── GLOBAL SCOPE
    ├── f
    └── main
        ├── argc
        ├── argv
        ├── x = 2
        └── f(x)
            ├── y = 2
            ├── x = 3
            ├── for-loop (Iteration 0)
            │   ├── i = 0
            │   └── temp = 3
            ├── for-loop (Iteration 1)
            │   ├── i = 1
            │   └── temp = 5
            └── for-loop (Iteration 2)
                ├── i = 2
                └── temp = 7

#include <iostream>
int main() {
    int* p = new int(42);
    {
        int* q = p;
        std::cout << *q << std::endl;
    }
    delete p;
}

OUTLINE:
.
└── GLOBAL SCOPE
    └── main
        ├── p (int* stack pointer)
        ├── Inner Block Scope
        │   └── q (int* stack pointer -> points to p heap memory)
        └── HEAP STORAGE
            └── int(42) (Dynamic Memory)
            └── delete p
